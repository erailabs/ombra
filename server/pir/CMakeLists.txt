# Private record retrieval (BFV PIR), server side.
#   ombra_pir_layout  plain C++: BLAKE2b, cuckoo hashing, call shape, row -> coefficient packing.
#                     No OpenFHE, so the router can link it to size a call.
#   ombra_pir_answer  OpenFHE on the CPU: context and key checks, expansion, the answer. Built when
#                     OpenFHE 1.6.1 is installed (always in a CPU build, and in a CUDA build).

# ---- The retrieval preset, generated from the preset JSON (no hand-picked numbers) ----------
get_filename_component(OMBRA_PRESET_JSON ${CMAKE_CURRENT_SOURCE_DIR}/../../core/params/ckks_presets.json ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${OMBRA_PRESET_JSON})
file(READ ${OMBRA_PRESET_JSON} _pj)
set(_pir_name pir_bfv_n8192)
string(JSON _scheme GET ${_pj} presets ${_pir_name} scheme)
string(JSON _sec GET ${_pj} presets ${_pir_name} security_level)
string(JSON _ring GET ${_pj} presets ${_pir_name} ring_dim)
string(JSON _depth GET ${_pj} presets ${_pir_name} multiplicative_depth)
string(JSON _t GET ${_pj} presets ${_pir_name} plaintext_modulus)
string(JSON _ks GET ${_pj} presets ${_pir_name} key_switch_technique)
string(JSON _digit GET ${_pj} presets ${_pir_name} digit_size)
if(NOT _scheme STREQUAL "BFV")
  message(FATAL_ERROR "preset ${_pir_name} must be a BFV preset")
endif()
if(NOT _sec MATCHES "^HEStd_(128|192|256)_(classic|quantum)$")
  message(FATAL_ERROR "preset ${_pir_name}: security level ${_sec} is below 128 bits")
endif()
if(NOT _ks MATCHES "^(BV|HYBRID)$")
  message(FATAL_ERROR "preset ${_pir_name}: unknown key switch technique ${_ks}")
endif()
set(_gen ${CMAKE_CURRENT_BINARY_DIR}/gen)
file(WRITE ${_gen}/ombra/pir/preset.hpp.tmp
"// Generated by CMake from core/params/ckks_presets.json. Do not edit.
#pragma once
#include <cstdint>
namespace ombra::pir::preset {
inline constexpr const char* kName = \"${_pir_name}\";
inline constexpr const char* kSecurityLevel = \"${_sec}\";
inline constexpr std::uint32_t kRingDim = ${_ring};
inline constexpr std::uint32_t kMultiplicativeDepth = ${_depth};
inline constexpr std::uint64_t kPlaintextModulus = ${_t};
inline constexpr const char* kKeySwitchTechnique = \"${_ks}\";
inline constexpr std::uint32_t kDigitSize = ${_digit};
}  // namespace ombra::pir::preset
")
file(COPY_FILE ${_gen}/ombra/pir/preset.hpp.tmp ${_gen}/ombra/pir/preset.hpp ONLY_IF_DIFFERENT)

add_library(ombra_pir_layout STATIC src/blake2b.cpp src/layout.cpp)
target_include_directories(ombra_pir_layout PUBLIC include ${_gen})

add_executable(pir_layout_test tests/layout_test.cpp)
target_link_libraries(pir_layout_test PRIVATE ombra_pir_layout)
target_compile_definitions(pir_layout_test
                           PRIVATE OMBRA_PIR_GOLDEN_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/golden")
add_test(NAME pir_layout COMMAND pir_layout_test)

# ---- The answer, on OpenFHE (CPU) ------------------------------------------------------------
find_package(OpenFHE 1.6.1 EXACT QUIET)
if(NOT OpenFHE_FOUND)
  message(STATUS "ombra_pir_answer not built: OpenFHE 1.6.1 not found")
  return()
endif()
if(NOT "${OpenFHE_NATIVE_SIZE}" STREQUAL "64")
  message(FATAL_ERROR "OpenFHE must be built with NATIVE_SIZE=64 (found '${OpenFHE_NATIVE_SIZE}')")
endif()
string(REPLACE "-Werror" "" _ofhe_flags "${OpenFHE_CXX_FLAGS}")
separate_arguments(_ofhe_flags UNIX_COMMAND "${_ofhe_flags}")
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  list(FILTER _ofhe_flags EXCLUDE REGEX "^-fopenmp$")
endif()
set(_core ${CMAKE_CURRENT_SOURCE_DIR}/../../core)
if(OMBRA_ENABLE_TEST_HOOKS)
  set_source_files_properties(${_core}/src/security_floor.cpp PROPERTIES COMPILE_DEFINITIONS
                                                                        OMBRA_ENABLE_TEST_HOOKS)
endif()
# The security floor of core/ is plain C++ over OpenFHE; compiled here so no CUDA library is needed
# (the CPU search library does the same). A CUDA build links ombra_core, which has it too: the
# object is compiled once per library, and only this one is linked into the answer.
add_library(ombra_pir_answer STATIC src/answer.cpp ${_core}/src/security_floor.cpp)
target_include_directories(
  ombra_pir_answer PUBLIC include ${_core}/include ${OpenFHE_INCLUDE}
                          ${OpenFHE_INCLUDE}/third-party/include ${OpenFHE_INCLUDE}/core
                          ${OpenFHE_INCLUDE}/pke ${OpenFHE_INCLUDE}/binfhe)
target_compile_options(ombra_pir_answer PRIVATE ${_ofhe_flags})
target_link_directories(ombra_pir_answer PUBLIC ${OpenFHE_LIBDIR})
target_link_libraries(ombra_pir_answer PUBLIC ombra_pir_layout ${OpenFHE_SHARED_LIBRARIES})

add_executable(pir_answer_test tests/answer_test.cpp)
target_compile_options(pir_answer_test PRIVATE ${_ofhe_flags})
target_link_libraries(pir_answer_test PRIVATE ombra_pir_answer)
add_test(NAME pir_answer COMMAND pir_answer_test)
