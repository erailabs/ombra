# Search engine library. The pure logic (layout, tombstone penalty vector) is plain C++ and
# tested on the CPU. The engine itself needs OpenFHE and CUDA and is built with
# OMBRA_ENABLE_CUDA=ON. With OMBRA_ENABLE_CUDA=OFF and OpenFHE installed, a CPU-only engine is built
# instead (same API, same key checks, Search fails with a clear error): for tests and development.
add_library(ombra_search_logic STATIC src/layout.cpp)
target_include_directories(ombra_search_logic PUBLIC include)
target_link_libraries(ombra_search_logic PUBLIC ombra_storage)

add_executable(search_layout_test tests/layout_test.cpp)
target_link_libraries(search_layout_test PRIVATE ombra_search_logic)
add_test(NAME search_layout COMMAND search_layout_test)

# ---- Full preset parameters, generated from the preset JSON (no hand-picked numbers) ---------
# key_prep.cpp builds each preset's crypto context from this table, with the same settings as
# client/ombra/params.py, and requires an uploaded context to equal it byte for byte. The context
# the Python client writes differs from the C++ writer's in a few class-version words (see
# context_preflight.hpp), so the context as the client writes it is embedded too, from
# context_golden/<preset>.bin (made by context_golden/generate.py with the real client).
set(OMBRA_PRESET_JSON ${CMAKE_CURRENT_SOURCE_DIR}/../../core/params/ckks_presets.json)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${OMBRA_PRESET_JSON})
file(READ ${OMBRA_PRESET_JSON} _pj)
string(JSON _pcount LENGTH ${_pj} presets)
math(EXPR _plast "${_pcount} - 1")
set(_pentries "")
set(_parrays "")
foreach(_i RANGE ${_plast})
  string(JSON _n MEMBER ${_pj} presets ${_i})
  string(JSON _sec GET ${_pj} presets ${_n} security_level)
  if(NOT _sec MATCHES "^HEStd_(128|192|256)_(classic|quantum)$")
    message(FATAL_ERROR "preset ${_n}: security level ${_sec} is below 128 bits")
  endif()
  string(JSON _ring GET ${_pj} presets ${_n} ring_dim)
  string(JSON _depth GET ${_pj} presets ${_n} multiplicative_depth)
  string(JSON _scale GET ${_pj} presets ${_n} scaling_mod_size)
  string(JSON _first GET ${_pj} presets ${_n} first_mod_size)
  string(JSON _batch GET ${_pj} presets ${_n} batch_size)
  string(JSON _st GET ${_pj} presets ${_n} scaling_technique)
  string(JSON _ks GET ${_pj} presets ${_n} key_switch_technique)
  if(NOT _st MATCHES "^(FIXEDMANUAL|FIXEDAUTO|FLEXIBLEAUTO|FLEXIBLEAUTOEXT)$")
    message(FATAL_ERROR "preset ${_n}: unknown scaling technique ${_st}")
  endif()
  if(NOT _ks MATCHES "^(BV|HYBRID)$")
    message(FATAL_ERROR "preset ${_n}: unknown key switch technique ${_ks}")
  endif()
  string(JSON _digits ERROR_VARIABLE _err GET ${_pj} presets ${_n} num_large_digits)
  if(_err)
    set(_digits 0)  # absent: OpenFHE picks its default, as the client does
  endif()
  set(_golden ${CMAKE_CURRENT_SOURCE_DIR}/context_golden/${_n}.bin)
  if(NOT EXISTS ${_golden})
    message(FATAL_ERROR "missing ${_golden}: run server/search/context_golden/generate.py")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_golden})
  file(READ ${_golden} _hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _hex "${_hex}")
  string(APPEND _parrays "inline constexpr unsigned char kClientContext_${_n}[] = {${_hex}};\n")
  string(APPEND _pentries
         "    {\"${_n}\", \"${_sec}\", ${_ring}, ${_depth}, ${_scale}, ${_first}, ${_batch}, \"${_st}\", \"${_ks}\", ${_digits}, kClientContext_${_n}, sizeof(kClientContext_${_n})},\n")
endforeach()
set(_pgen ${CMAKE_CURRENT_BINARY_DIR}/gen)
file(WRITE ${_pgen}/ckks_preset_params.hpp.tmp
"// Generated by CMake from core/params/ckks_presets.json. Do not edit.
#pragma once
#include \"context_preflight.hpp\"
namespace ombra::search::detail::generated {
${_parrays}inline constexpr PresetParams kPresetParams[] = {
${_pentries}};
}  // namespace ombra::search::detail::generated
")
file(COPY_FILE ${_pgen}/ckks_preset_params.hpp.tmp ${_pgen}/ckks_preset_params.hpp ONLY_IF_DIFFERENT)

if(NOT OMBRA_ENABLE_CUDA)
  find_package(OpenFHE 1.6.1 EXACT QUIET)
  if(NOT OpenFHE_FOUND)
    message(STATUS "CPU-only search engine not built: OpenFHE 1.6.1 not found")
    return()
  endif()
  if(NOT "${OpenFHE_NATIVE_SIZE}" STREQUAL "64")
    message(FATAL_ERROR "OpenFHE must be built with NATIVE_SIZE=64 (found '${OpenFHE_NATIVE_SIZE}')")
  endif()
  # Same flags as core/CMakeLists.txt: OpenFHE's backend defines and OpenMP, without -Werror.
  string(REPLACE "-Werror" "" _ofhe_flags "${OpenFHE_CXX_FLAGS}")
  separate_arguments(_ofhe_flags UNIX_COMMAND "${_ofhe_flags}")
  if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    # The fuzz build uses clang, whose OpenMP runtime does not match OpenFHE's (GCC, libgomp).
    list(FILTER _ofhe_flags EXCLUDE REGEX "^-fopenmp$")
  endif()
  set(_core ${CMAKE_CURRENT_SOURCE_DIR}/../../core)
  if(OMBRA_ENABLE_TEST_HOOKS)
    set_source_files_properties(${_core}/src/security_floor.cpp PROPERTIES COMPILE_DEFINITIONS
                                                                          OMBRA_ENABLE_TEST_HOOKS)
  endif()
  # The security floor of core/ is plain C++ over OpenFHE: compiled here, no CUDA library needed.
  add_library(ombra_search STATIC src/session_cpu.cpp src/key_prep.cpp
                                  src/context_preflight.cpp ${_core}/src/security_floor.cpp)
  target_include_directories(
    ombra_search PUBLIC include ${_core}/include ${OpenFHE_INCLUDE}
                        ${OpenFHE_INCLUDE}/third-party/include ${OpenFHE_INCLUDE}/core
                        ${OpenFHE_INCLUDE}/pke ${OpenFHE_INCLUDE}/binfhe PRIVATE src ${_pgen})
  target_compile_options(ombra_search PRIVATE ${_ofhe_flags})
  target_link_directories(ombra_search PUBLIC ${OpenFHE_LIBDIR})
  target_link_libraries(ombra_search PUBLIC ombra_search_logic ombra_cache_logic ombra_storage
                                            ${OpenFHE_SHARED_LIBRARIES})

  # Preflight of context and key files; includes the private headers of src and the generated table.
  add_executable(search_context_preflight_test tests/context_preflight_test.cpp)
  target_include_directories(search_context_preflight_test PRIVATE src ${_pgen})
  target_compile_options(search_context_preflight_test PRIVATE ${_ofhe_flags})
  target_compile_definitions(search_context_preflight_test
                             PRIVATE OMBRA_GOLDEN_DIR="${CMAKE_CURRENT_SOURCE_DIR}/context_golden")
  target_link_libraries(search_context_preflight_test PRIVATE ombra_search)
  add_test(NAME search_context_preflight COMMAND search_context_preflight_test)
  return()
endif()

add_library(ombra_search STATIC src/session.cpp src/key_prep.cpp src/context_preflight.cpp
                                src/device_util.cu)
target_include_directories(ombra_search PUBLIC include PRIVATE src ${_pgen})
set_target_properties(ombra_search PROPERTIES CUDA_SEPARABLE_COMPILATION OFF CUDA_STANDARD 17)
target_link_libraries(ombra_search PUBLIC ombra_search_logic ombra_cache ombra_core ombra_storage)
